## 2. Layer 1: RegisteredToolFunction — 注册元数据层

### 2.1 当前状态

.NET 的 `Toolkit` 直接用 `Dictionary<string, ITool>` 存储工具，注册时无法附加任何元信息：

```csharp
// 当前: 只能存工具实例，信息丢失
_toolkit.AddTool(new CalculatorTool());
```

### 2.2 Java 参考

```java
// Java: RegisteredToolFunction 包装三层元数据
class RegisteredToolFunction {
    AgentTool tool;                    // 工具实例
    ExtendedModel extendedModel;       // Schema 扩展 (可能为 null)
    String mcpClientName;             // MCP 来源 (非 MCP 工具为 null)
    Map<String, Object> presetParameters; // 预设参数 (运行时注入)
    
    // 合并 base + extension Schema
    Map<String, Object> getExtendedParameters();
}

// Toolkit 内部的 ToolRegistry 维护双表
class ToolRegistry {
    Map<String, AgentTool> tools;                // 工具实例
    Map<String, RegisteredToolFunction> registeredTools; // 元数据
    
    void registerTool(String name, AgentTool tool, RegisteredToolFunction registered);
    AgentTool getTool(String name);
    RegisteredToolFunction getRegisteredTool(String name);
}
```

### 2.3 .NET 实施方案

#### 2.3.1 新建文件 `RegisteredToolFunction.cs`

```csharp
namespace AgentScope.Core.Tool;

/// <summary>
/// AgentTool 的元数据包装器。
/// 每个 RegisteredToolFunction 关联一个工具实例及其扩展信息：
/// - ExtendedModel: 运行时动态注入的参数 Schema
/// - McpClientName: 来源 MCP 客户端名 (非 MCP 工具为 null)
/// - PresetParameters: 预设参数值 (框架注入，LLM 不可修改)
/// </summary>
public class RegisteredToolFunction
{
    public ITool Tool { get; }
    public ExtendedModel? ExtendedModel { get; }
    public string? McpClientName { get; }
    public IReadOnlyDictionary<string, object> PresetParameters { get; private set; }

    public RegisteredToolFunction(
        ITool tool,
        ExtendedModel? extendedModel = null,
        string? mcpClientName = null,
        IDictionary<string, object>? presetParameters = null)
    {
        Tool = tool ?? throw new ArgumentNullException(nameof(tool));
        ExtendedModel = extendedModel;
        McpClientName = mcpClientName;
        PresetParameters = presetParameters?.AsReadOnly()
            ?? System.Collections.Immutable.ImmutableDictionary<string, object>.Empty;
    }

    /// <summary>合并 base Schema 与 ExtendedModel，返回最终参数 Schema。</summary>
    public Dictionary<string, object> GetExtendedParameters()
    {
        var baseParams = Tool.GetSchema();
        if (ExtendedModel == null) return baseParams;
        return ExtendedModel.MergeWithBaseSchema(baseParams);
    }

    internal void UpdatePresetParameters(IDictionary<string, object>? newParams)
    {
        PresetParameters = newParams?.AsReadOnly()
            ?? System.Collections.Immutable.ImmutableDictionary<string, object>.Empty;
    }
}
```

#### 2.3.2 新建文件 `ToolRegistry.cs`

```csharp
namespace AgentScope.Core.Tool;

/// <summary>
/// 工具注册表，维护工具实例与元数据的双层存储。
/// 线程安全 (ConcurrentDictionary)。
/// </summary>
internal class ToolRegistry
{
    private readonly ConcurrentDictionary<string, ITool> _tools = new(StringComparer.OrdinalIgnoreCase);
    private readonly ConcurrentDictionary<string, RegisteredToolFunction> _registeredTools
        = new(StringComparer.OrdinalIgnoreCase);

    public void RegisterTool(string name, ITool tool, RegisteredToolFunction? registered = null)
    {
        _tools[name] = tool;
        if (registered != null) _registeredTools[name] = registered;
    }

    public ITool? GetTool(string name)
        => _tools.TryGetValue(name ?? "", out var t) ? t : null;

    public RegisteredToolFunction? GetRegisteredTool(string name)
        => _registeredTools.TryGetValue(name ?? "", out var r) ? r : null;

    public IReadOnlyCollection<string> GetToolNames()
        => _tools.Keys.ToList();

    public IReadOnlyDictionary<string, RegisteredToolFunction> GetAllRegisteredTools()
        => new Dictionary<string, RegisteredToolFunction>(_registeredTools);

    public void RemoveTool(string name)
    {
        _tools.TryRemove(name ?? "", out _);
        _registeredTools.TryRemove(name ?? "", out _);
    }

    public bool RemoveToolIfSame(string name, ITool expected)
    {
        // CAS 式删除：只有当前实例与 expected 一致才删除
        var removed = _tools.TryRemove(KeyValuePair.Create(name ?? "", expected));
        if (removed) _registeredTools.TryRemove(name ?? "", out _);
        return removed;
    }

    public void CopyTo(ToolRegistry target)
    {
        foreach (var kv in _tools)
        {
            var reg = _registeredTools.GetValueOrDefault(kv.Key);
            target.RegisterTool(kv.Key, kv.Value, reg);
        }
    }
}
```

#### 3.3.3 修改 `ToolGroupManager.cs`

关键新增方法：

```csharp
public class ToolGroupManager
{
    private readonly Dictionary<string, ToolGroup> _groups = new(StringComparer.OrdinalIgnoreCase);

    // ---------- 注册 ----------

    public void RegisterGroup(ToolGroup group) { /* 不变 */ }

    // ---------- 范围感知查询 ----------

    /// <summary>获取所有 META 范围组的名称。</summary>
    public HashSet<string> GetMetaGroupNames()
        => _groups.Values.Where(g => g.Scope == ToolGroupScope.Meta)
                 .Select(g => g.Name).ToHashSet(StringComparer.OrdinalIgnoreCase);

    /// <summary>获取 META 范围组的状态备注 (供 reset_equipped_tools 注入)。</summary>
    public string GetNotes()
    {
        var metaGroups = _groups.Values.Where(g => g.Scope == ToolGroupScope.Meta);
        var active = new List<string>();
        var inactive = new List<string>();
        foreach (var g in metaGroups)
        {
            if (g.IsActive) active.Add($"- {g.Name}: {g.Description}");
            else inactive.Add($"- {g.Name}: {g.Description}");
        }
        var sb = new StringBuilder();
        sb.AppendLine("Activated tool groups:");
        sb.AppendLine(active.Count > 0 ? string.Join("\n", active) : "  (none)");
        sb.AppendLine();
        sb.AppendLine("Inactive tool groups:");
        sb.AppendLine(inactive.Count > 0 ? string.Join("\n", inactive) : "  (none)");
        return sb.ToString();
    }

    /// <summary>替换语义：停用所有 META 组，仅激活指定的组。External 组不受影响。</summary>
    public void ReplaceMetaActiveGroups(IEnumerable<string> toActivate)
    {
        var activateSet = toActivate?.ToHashSet(StringComparer.OrdinalIgnoreCase)
                          ?? new HashSet<string>();

        foreach (var group in _groups.Values)
        {
            if (group.Scope == ToolGroupScope.Meta && group.IsActive)
                group.IsActive = false;
        }
        foreach (var name in activateSet)
        {
            if (_groups.TryGetValue(name, out var g) && g.Scope == ToolGroupScope.Meta)
                g.IsActive = true;
        }
    }

    // ---------- 工具激活检测 (增强) ----------

    /// <summary>
    /// 检查工具是否在任一激活组中。
    /// 未分组的工具始终视为激活 (保持向后兼容)。
    /// </summary>
    public bool IsActiveTool(string toolName)
    {
        if (string.IsNullOrEmpty(toolName)) return false;
        // 查找工具所属的组 (需要工具-组反向索引)
        var groups = GetGroupsForTool(toolName);
        if (groups.Count == 0) return true;  // 未分组的工具默认激活
        return groups.Any(g => g.IsActive);
    }

    /// <summary>检查工具是否已被分组。</summary>
    public bool IsGroupedTool(string toolName)
        => GetGroupsForTool(toolName).Count > 0;

    // ---------- 反向索引 ----------

    private readonly Dictionary<string, HashSet<string>> _toolToGroups
        = new(StringComparer.OrdinalIgnoreCase);

    private List<ToolGroup> GetGroupsForTool(string toolName)
    {
        if (!_toolToGroups.TryGetValue(toolName ?? "", out var groupNames))
            return new List<ToolGroup>();
        return groupNames.Select(n => _groups.GetValueOrDefault(n))
                         .Where(g => g != null).Cast<ToolGroup>().ToList();
    }

    public void AddToolToGroup(string groupName, string toolName)
    {
        if (_groups.TryGetValue(groupName ?? "", out var group))
        {
            group.AddTool(toolName);
            _toolToGroups.GetOrAdd(toolName ?? "",
                _ => new HashSet<string>(StringComparer.OrdinalIgnoreCase)).Add(groupName);
        }
    }

    public void RemoveToolFromGroup(string groupName, string toolName)
    {
        if (_groups.TryGetValue(groupName ?? "", out var group))
        {
            group.RemoveTool(toolName);
            if (_toolToGroups.TryGetValue(toolName ?? "", out var groups))
            {
                groups.Remove(groupName);
                if (groups.Count == 0) _toolToGroups.Remove(toolName ?? "");
            }
        }
    }

    // ---------- 无状态变体 (用于 ReActAgent 多 slot 独立解析) ----------

    public HashSet<string> GetActiveToolNames()
        => _groups.Values.Where(g => g.IsActive)
                 .SelectMany(g => g.GetTools())
                 .ToHashSet(StringComparer.OrdinalIgnoreCase);

    public HashSet<string> GetActiveToolNames(ICollection<string>? activeGroups)
    {
        if (activeGroups == null) return new HashSet<string>();
        return _groups.Values.Where(g => activeGroups.Contains(g.Name) && g.IsActive)
                     .SelectMany(g => g.GetTools())
                     .ToHashSet(StringComparer.OrdinalIgnoreCase);
    }

    // ---------- 复制 ----------

    public ToolGroupManager Copy()
    {
        var copy = new ToolGroupManager();
        foreach (var kv in _groups)
            copy._groups[kv.Key] = new ToolGroup(kv.Value.Name, kv.Value.Description,
                                                  kv.Value.IsActive, kv.Value.Scope);
        foreach (var kv in _toolToGroups)
            copy._toolToGroups[kv.Key] = new HashSet<string>(kv.Value, StringComparer.OrdinalIgnoreCase);
        return copy;
    }
}
```

## 4. Layer 3: SchemaOnlyTool — 外部工具占位层

### 4.1 当前状态

.NET 中没有"只注册 Schema 不执行"的工具概念。所有 `ITool` 必须有 `ExecuteAsync` 实现。

### 4.2 Java 参考

```java
// Java SchemaOnlyTool: 只有 Schema, 无执行逻辑
public class SchemaOnlyTool extends ToolBase {
    public SchemaOnlyTool(String name, String description, Map<String, Object> parameters) {
        super(ToolBase.builder()
                .name(name).description(description)
                .inputSchema(parameters).externalTool(true)
                .readOnly(false).concurrencySafe(true));
    }

    @Override
    public Mono<ToolResultBlock> callAsync(ToolCallParam param) {
        return Mono.error(new ToolSuspendException());   // 备用防御
    }
}

// ToolExecutor.executeCore() 提前短路:
if (tool instanceof ToolBase tb && tb.isExternalTool()) {
    return Mono.just(ToolResultBlock.suspended(toolCall));  // 不调 callAsync
}
```

关键语义：
- `externalTool = true` → 框架在 ToolExecutor 中短路，不调 `callAsync`
- 直接返回 `ToolResultBlock.suspended()` → Agent 收到 `TOOL_SUSPENDED`，将控制权交还给用户/外部系统
- `callAsync` 中的 `ToolSuspendException` 是备用防御

### 4.3 .NET 实施方案

#### 4.3.1 修改 `ITool.cs` — 增加 IsExternal 属性

```csharp
{
    string Name { get; }
    string Description { get; }
    Dictionary<string, object> GetSchema();

    /// <summary>
    /// 是否为外部工具。
    /// 外部工具只有 Schema 没有执行逻辑，调用时框架直接返回 suspension 信号。
    /// </summary>
```

#### 4.3.2 新建文件 `SchemaOnlyTool.cs`

```csharp
namespace AgentScope.Core.Tool;

/// <summary>
/// 仅有 Schema 定义没有执行逻辑的外部工具占位。
///
/// 当 Model 调用 SchemaOnlyTool 时，框架将其标记为 "suspended"，
/// 等待外部系统执行并返回结果。Agent 收到 suspension 信号后暂停 ReAct 循环。
///
/// 对标: Java io.agentscope.core.tool.SchemaOnlyTool
/// </summary>
public class SchemaOnlyTool : ToolBase
{
    private readonly Dictionary<string, object> _schema;

    public SchemaOnlyTool(string name, string description, Dictionary<string, object> parameters)
        : base(name, description)
    {
        _schema = new Dictionary<string, object>
        {
            ["name"] = name,
            ["description"] = description,
            ["parameters"] = parameters ?? new Dictionary<string, object>()
        };
    }

    public SchemaOnlyTool(string name, string description, Dictionary<string, object> parameters, bool strict)
        : this(name, description, parameters)
    {
        _schema["strict"] = strict;
    }

    /// <summary>外部工具：永远为 true。</summary>
    public override bool IsExternal => true;

    public override Dictionary<string, object> GetSchema() => _schema;

    /// <summary>
    /// 不应被直接调用。ToolExecutor 会在 IsExternal=true 时短路返回 suspended。
    /// 此方法作为备用防御。
    /// </summary>
    public override Task<ToolResult> ExecuteAsync(Dictionary<string, object> parameters)
    {
        throw new ToolSuspendException(
            $"SchemaOnlyTool '{Name}' is external and cannot be executed locally.");
    }
}
```

#### 4.3.3 修改 `ToolBase.cs`

```csharp
public abstract class ToolBase : ITool
{
    public string Name { get; protected set; }
    public string Description { get; protected set; }

    /// <summary>
```

```csharp
namespace AgentScope.Core.Tool;

/// <summary>
/// 向工具参数 Schema 添加额外属性的扩展模型接口。
/// 用于 MCP 工具注册、远程工具 Schema 合并等场景。
///
/// 对标: Java io.agentscope.core.tool.ExtendedModel
/// </summary>
public interface ExtendedModel
{
    /// <summary>额外属性键值对。</summary>
    IReadOnlyDictionary<string, object> AdditionalProperties { get; }

    /// <summary>额外必填字段列表。</summary>
    IReadOnlyList<string> AdditionalRequired { get; }

    /// <summary>
    /// 合并此扩展模型与工具的 base Schema。
    /// 检测属性冲突，如果扩展属性与 base 属性重名则抛异常。
    /// </summary>
    Dictionary<string, object> MergeWithBaseSchema(Dictionary<string, object> baseSchema)
    {
        // 提取 base 中的 parameters 子结构
        if (!baseSchema.TryGetValue("parameters", out var paramsObj) ||
            paramsObj is not Dictionary<string, object> parameters)
        {
            // 直接返回扩展参数作为 parameters
            var result = new Dictionary<string, object>(baseSchema);
            result["parameters"] = BuildExtendedParametersSchema();
            return result;
        }

        var mergedParams = new Dictionary<string, object>(parameters);

        // properties 合并
        var baseProps = parameters.TryGetValue("properties", out var bp) && bp is Dictionary<string, object> basePropsDict
            ? new Dictionary<string, object>(basePropsDict)
            : new Dictionary<string, object>();

        var conflicts = AdditionalProperties.Keys.Where(k => baseProps.ContainsKey(k)).ToList();
        if (conflicts.Count > 0)
            throw new InvalidOperationException(
                $"Extended model has conflicting properties: {string.Join(", ", conflicts)}");

        foreach (var kv in AdditionalProperties)
            baseProps[kv.Key] = kv.Value;

        mergedParams["properties"] = baseProps;

        // required 合并
        var baseRequired = parameters.TryGetValue("required", out var br) && br is IList<object> baseReqList
            ? baseReqList.Select(r => r?.ToString()).Where(r => r != null).Cast<string>().ToHashSet()
            : new HashSet<string>();

        foreach (var r in AdditionalRequired)
            baseRequired.Add(r);

        if (baseRequired.Count > 0)
            mergedParams["required"] = baseRequired.ToList();

        var resultSchema = new Dictionary<string, object>(baseSchema);
        resultSchema["parameters"] = mergedParams;
        return resultSchema;
    }

    private Dictionary<string, object> BuildExtendedParametersSchema()
    {
        return new Dictionary<string, object>
        {
            ["type"] = "object",
            ["properties"] = new Dictionary<string, object>(AdditionalProperties),
            ["required"] = AdditionalRequired.ToList()
        };
    }
}
```

### 6.1 当前状态

### 6.3 .NET 实施方案

#### 6.3.1 新建文件 `MetaToolFactory.cs`

```csharp
namespace AgentScope.Core.Tool;

/// <summary>
/// 创建元工具 (meta tool)，让 Agent 在运行时动态管理 META 范围工具组的激活状态。
///
/// 核心工具: reset_equipped_tools
/// - Schema 中只包含 ToolGroupScope.Meta 范围的组
/// - 使用替换语义: 只保留 to_activate 中的组激活
///
/// 对标: Java io.agentscope.core.tool.MetaToolFactory
/// </summary>
internal class MetaToolFactory
{
    private readonly ToolGroupManager _groupManager;

    internal MetaToolFactory(ToolGroupManager groupManager)
    {
        _groupManager = groupManager;
    }

    /// <summary>创建 reset_equipped_tools 元工具。</summary>
    public ITool CreateResetEquippedToolsTool()
    {
        return new ResetEquippedToolsTool(_groupManager);
    }

    private class ResetEquippedToolsTool : ToolBase
    {
        private readonly ToolGroupManager _groupManager;
        private readonly Dictionary<string, object> _schema;

        public ResetEquippedToolsTool(ToolGroupManager groupManager)
            : base("reset_equipped_tools", BuildDescription(groupManager))
        {
            _groupManager = groupManager;
            _schema = BuildSchema(groupManager);
        }

        public override bool IsExternal => false;

        public override Dictionary<string, object> GetSchema() => _schema;

        public override Task<ToolResult> ExecuteAsync(Dictionary<string, object> parameters)
        {
            try
            {
                // 解析 to_activate 参数
                var toActivate = new List<string>();
                if (parameters.TryGetValue("to_activate", out var raw))
                {
                    if (raw is IEnumerable<string> strs)
                        toActivate.AddRange(strs);
                    else if (raw is System.Text.Json.JsonElement je)
                    {
                        // 处理 JSON 反序列化场景
                        var arr = System.Text.Json.JsonSerializer.Deserialize<List<string>>(je.GetRawText());
                        if (arr != null) toActivate.AddRange(arr);
                    }
                }

                // 验证：所有组必须存在且为 Meta 范围
                foreach (var name in toActivate)
                {
                    var group = _groupManager.GetToolGroup(name);
                    if (group == null)
                        return Task.FromResult(ToolResult.Fail(
                            $"Tool group '{name}' does not exist."));
                    if (group.Scope != ToolGroupScope.Meta)
                        return Task.FromResult(ToolResult.Fail(
                            $"Group '{name}' is not manageable by this tool (scope={group.Scope})."));
                }

                // 替换语义：停用所有 Meta 组，激活指定的
                _groupManager.ReplaceMetaActiveGroups(toActivate);

                // 构建响应
                var result = BuildResultMessage(toActivate);
                return Task.FromResult(ToolResult.Ok(result));
            }
            catch (Exception ex)
            {
                return Task.FromResult(ToolResult.Fail(ex.Message));
            }
        }

        private string BuildResultMessage(List<string> toActivate)
        {
            if (toActivate.Count == 0)
                return "All tool groups are currently deactivated.";

            var sb = new StringBuilder();
            sb.AppendLine($"The currently activated tool group(s): {string.Join(", ", toActivate)}.");
            sb.AppendLine();

            // 收集有描述的组，生成 <tool-instructions> 块
            var activatedGroups = toActivate
                .Select(n => _groupManager.GetToolGroup(n))
                .Where(g => g != null)
                .Cast<ToolGroup>()
                .Where(g => !string.IsNullOrEmpty(g.Description))
                .ToList();

            if (activatedGroups.Count > 0)
            {
                sb.AppendLine("<tool-instructions>");
                foreach (var g in activatedGroups)
                {
                    sb.AppendLine($"<group name=\"{g.Name}\">{g.Description}</group>");
                }
                sb.AppendLine("</tool-instructions>");
            }

            return sb.ToString();
        }

        private static string BuildDescription(ToolGroupManager groupManager)
        {
            return "Reset your equipped tools based on your current task requirements. ...\n\n"
                   + groupManager.GetNotes();
        }

        private static Dictionary<string, object> BuildSchema(ToolGroupManager groupManager)
        {
            var metaGroupNames = groupManager.GetMetaGroupNames().ToList();

            var items = new Dictionary<string, object>
            {
                ["type"] = "string"
            };
            if (metaGroupNames.Count > 0)
                items["enum"] = metaGroupNames;

            return new Dictionary<string, object>
            {
                ["name"] = "reset_equipped_tools",
                ["description"] = "Reset your equipped tools...",
                ["parameters"] = new Dictionary<string, object>
                {
                    ["type"] = "object",
                    ["properties"] = new Dictionary<string, object>
                    {
                        ["to_activate"] = new Dictionary<string, object>
                        {
                            ["type"] = "array",
                            ["items"] = items,
                            ["description"] = "The FINAL list of tool group names to keep active."
                        }
                    },
                    ["required"] = new List<string> { "to_activate" }
                }
            };
        }
    }
}
```

```csharp
{
```

```csharp
    private readonly List<IToolkitAware> _awareComponents = new();

    /// <summary>注册一个需要 Toolkit 重新绑定通知的组件。</summary>
    public Toolkit RegisterAwareComponent(IToolkitAware component)
    {
        if (component != null && !_awareComponents.Contains(component))
            _awareComponents.Add(component);
        return this;
    }

    /// <summary>
    /// 深拷贝 Toolkit。
    /// 拷贝后遍历 IToolkitAware 组件，通知它们绑定新实例。
    /// </summary>
    public Toolkit DeepCopy()
    {
        var copy = new Toolkit();

        // 1. 复制工具注册表
        _registry.CopyTo(copy._registry);

        // 2. 复制组管理器
        copy._groupManager = _groupManager.Copy();

        // 3. 复制 aware 组件 (但不清除原始组件的引用)
        copy._awareComponents.AddRange(_awareComponents);

        // 4. 通知所有 aware 组件绑定新实例
        foreach (var aware in _awareComponents)
        {
            try { aware.RebindToolkit(copy); }
            catch (Exception ex)
            {
                // 日志记录但不中断拷贝
                System.Diagnostics.Debug.WriteLine(
                    $"ToolkitAware.RebindToolkit failed: {ex.Message}");
            }
        }

        return copy;
    }

    // 旧 Copy() 方法保留作向后兼容，但内部调用 DeepCopy
    public Toolkit Copy() => DeepCopy();
}
```

## 8. Layer 7: ToolExecutor 调度管线增强 — 并行/外部工具调度

### 8.1 当前状态

.NET 的 `ToolExecutor` 是简单循环 + try-catch。`Toolkit.CallToolsAsync()` 是纯串行执行，无并行支持，无外部工具短路。

### 8.2 Java 参考

```java
// 并行模式 + concurrencySafe 分区:
Mono<List<ToolResultBlock>> executeAll(
    List<ToolUseBlock> toolCalls,
    boolean parallel,
    ExecutionConfig executionConfig,
    ...
) {
    if (!parallel) return Flux.concat(monos).collectList();  // 串行

    // 并行: 安全工具 mergeSequential, 不安全工具各自串行槽
    List<Flux<ToolResultBlock>> chunks = new ArrayList<>();
    List<Mono<ToolResultBlock>> safeBatch = new ArrayList<>();
    for (ToolUseBlock toolCall : toolCalls) {
        Mono<ToolResultBlock> mono = executeWithInfrastructure(toolCall, ...);
        if (isConcurrencySafe(toolCall)) {
            safeBatch.add(mono);
        } else {
            if (!safeBatch.isEmpty()) {
                chunks.add(Flux.mergeSequential(safeBatch));
                safeBatch = new ArrayList<>();
            }
            chunks.add(mono.flux());  // 不安全工具单一串行槽
        }
    }
    if (!safeBatch.isEmpty())
        chunks.add(Flux.mergeSequential(safeBatch));
    return Flux.concat(chunks).collectList();
}

// 调度管线:
private Mono<ToolResultBlock> executeWithInfrastructure(...) {
    Mono<ToolResultBlock> execution = execute(param);
    execution = applyScheduling(execution);     // 线程调度
    execution = applyTimeout(execution, ...);   // 超时控制
    execution = applyRetry(execution, ...);     // 重试+退避
    execution = applyShutdownGuard(execution);  // 优雅关闭保护
    return execution;
}

// 外部工具短路:
if (tool instanceof ToolBase tb && tb.isExternalTool()) {
    return Mono.just(ToolResultBlock.suspended(toolCall));
}
```

### 8.3 .NET 实施方案

#### 8.3.1 修改 `ToolExecutor.cs` — 增加批量执行

```csharp
public class ToolExecutor
{
    // 单工具执行 (不变)
    public async Task<ToolResult> ExecuteAsync(
        ITool tool, Dictionary<string, object> parameters,
        CancellationToken cancellationToken = default)
    {
        // for 循环 + 重试 + 超时 (不变)
    }

    // ==================== 批量执行 (新增) ====================

    /// <summary>
    /// 批量执行工具调用，支持并行/串行控制、超时、重试和外部工具短路。
    /// </summary>
    public async Task<List<ToolResultBlock>> ExecuteAllAsync(
        IReadOnlyList<ToolUseBlock> toolCalls,
        bool parallel,
        ExecutionConfig? config = null,
        CancellationToken cancellationToken = default)
    {
        if (toolCalls == null || toolCalls.Count == 0)
            return new List<ToolResultBlock>();

        // 串行模式
        if (!parallel)
        {
            var results = new List<ToolResultBlock>();
            foreach (var call in toolCalls)
            {
                var result = await ExecuteSingleWithInfrastructure(
                    call, config, cancellationToken);
                if (result != null) results.Add(result);
            }
            return results;
        }

        // 并行模式 + concurrencySafe 分区:
        // 安全工具批量并发, 不安全工具各自串行
        var batches = new List<List<MonoLike<ToolResultBlock?>>>();
        var currentSafe = new List<MonoLike<ToolResultBlock?>>();

        foreach (var call in toolCalls)
        {
            var mono = new MonoLike<ToolResultBlock?>(
                () => ExecuteSingleWithInfrastructure(call, config, cancellationToken));

            if (IsConcurrencySafe(call.Name))
            {
                currentSafe.Add(mono);
            }
            else
            {
                if (currentSafe.Count > 0)
                {
                    batches.Add(new List<MonoLike<ToolResultBlock?>>(currentSafe));
                    currentSafe.Clear();
                }
                // 不安全工具各自独立串行
                batches.Add(new List<MonoLike<ToolResultBlock?>> { mono });
            }
        }
        if (currentSafe.Count > 0)
            batches.Add(currentSafe);

        // 批次内并发, 批次间串行
        var allResults = new List<ToolResultBlock>();
        foreach (var batch in batches)
        {
            var batchTasks = batch.Select(m => m.Execute()).ToArray();
            var batchResults = await Task.WhenAll(batchTasks);
            foreach (var r in batchResults)
                if (r != null) allResults.Add(r);
        }
        return allResults;
    }

    // ==================== 单工具执行 + 基础设施 (新增) ====================

    private async Task<ToolResultBlock?> ExecuteSingleWithInfrastructure(
        ToolUseBlock call,
        ExecutionConfig? config,
        CancellationToken cancellationToken)
    {
        var tool = ResolveTool(call.Name);
        if (tool == null)
        {
            return new ToolResultBlock
            {
                Id = call.Id, Name = call.Name,
                IsError = true, Output = $"Unknown tool: {call.Name}"
            };
        }

        // 外部工具短路
        if (tool.IsExternal)
        {
            return new ToolResultBlock
            {
                Id = call.Id, Name = call.Name,
                IsError = false, IsSuspended = true,
                Output = "suspended"
            };
        }

        // Schema 验证
        var validationErrors = ToolValidator.Validate(tool.GetSchema(), call.Input);
        if (validationErrors.Count > 0)
        {
            return new ToolResultBlock
            {
                Id = call.Id, Name = call.Name,
                IsError = true,
                Output = $"Parameter validation failed: {string.Join("; ", validationErrors)}"
            };
        }

        try
        {
            var result = await ExecuteAsync(tool, call.Input ?? new(), cancellationToken);
            return new ToolResultBlock
            {
                Id = call.Id, Name = call.Name,
                IsError = !result.Success,
                IsSuspended = result.IsSuspended,
                Output = result.Success ? result.Result : result.Error
            };
        }
        catch (ToolSuspendException)
        {
            return new ToolResultBlock
            {
                Id = call.Id, Name = call.Name,
                IsError = false, IsSuspended = true,
                Output = "suspended"
            };
        }
        catch (OperationCanceledException)
        {
            return new ToolResultBlock
            {
                Id = call.Id, Name = call.Name,
                IsError = true, Output = "Tool execution cancelled"
            };
        }
        catch (Exception ex)
        {
            return new ToolResultBlock
            {
                Id = call.Id, Name = call.Name,
                IsError = true, Output = $"Tool execution failed: {ex.Message}"
            };
        }
    }

    private ITool? ResolveTool(string name) { /* 委托到 Toolkit 或注册表 */ }

    private bool IsConcurrencySafe(string toolName)
    {
        var tool = ResolveTool(toolName);
        if (tool is IToolConcurrencySafe safe) return safe.IsConcurrencySafe;
        return true; // 默认安全
    }

    /// <summary>简单的异步操作包装，支持并发执行。</summary>
    private class MonoLike<T>
    {
        private readonly Func<Task<T>> _factory;
        public MonoLike(Func<Task<T>> factory) => _factory = factory;
        public Task<T> Execute() => _factory();
    }
}
```

```csharp
namespace AgentScope.Core.Tool;

/// <summary>
/// 声明工具的并发安全性。
/// 并发安全的工具可与同批次的其他安全工具同时执行。
/// 不安全的工具始终在独立的串行槽中执行。
///
/// 对标: Java ToolBase.concurrencySafe 字段
/// </summary>
public interface IToolConcurrencySafe
{
    /// <summary>
    /// 是否可与此工具的另一个实例并发执行。
    /// true = 可并行；false = 必须串行 (例如文件写操作)。
    /// </summary>
    bool IsConcurrencySafe { get; }
}
```
